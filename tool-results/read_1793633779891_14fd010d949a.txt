   148→   148→export function resolvePalette(preset: string, overrides: Partial<RosterPalette> | undefined): RosterPalette {
   149→   149→  const table: Record<string, RosterPalette> = PRESETS
   150→   150→  const base = table[preset] || PRESETS.ember
   151→   151→  const out: RosterPalette = Object.assign({}, base)
   152→   152→  if (overrides) for (const k of Object.keys(base) as (keyof RosterPalette)[]) if (overrides[k]) out[k] = overrides[k]!
   153→   153→  return out
   154→   154→}

   156→   156→/** "#f80" or "#ff8800" to [r, g, b] in 0..1, for the SVG tritone tables. */
   157→   157→export function hexToUnit(hex: string) {
   158→   158→  let h = String(hex || "").trim().replace("#", "")
   159→   159→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
   160→   160→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return [0, 0, 0]
   161→   161→  return [0, 2, 4].map((i) => Math.round((parseInt(h.slice(i, i + 2), 16) / 255) * 1000) / 1000)
   162→   162→}
   163→   163→
   164→   164→/** Mixes two hex colours, t = 0 gives a, t = 1 gives b. */
   165→   165→export function mixHex(a: string, b: string, t: number) {
   166→   166→  const x = hexToUnit(a)
   167→   167→  const y = hexToUnit(b)
   168→   168→  return "#" + [0, 1, 2].map((i) => Math.round((x[i] + (y[i] - x[i]) * t) * 255).toString(16).padStart(2, "0")).join("")
   169→   169→}

   171→   171→/** One feFuncR/G/B table per channel: shadow, midtone, highlight. */
   172→   172→export function tritoneTables(shadow: string, mid: string, highlight: string) {
   173→   173→  const s = hexToUnit(shadow)
   174→   174→  const m = hexToUnit(mid)
   175→   175→  const h = hexToUnit(highlight)
   176→   176→  return [0, 1, 2].map((c) => s[c] + " " + m[c] + " " + h[c])
   177→   177→}

   179→   179→export function initials(name: string) {
   180→   180→  const parts = String(name || "").trim().split(/\s+/).filter(Boolean)
   181→   181→  if (!parts.length) return "?"
   182→   182→  const first = parts[0][0]
   183→   183→  const last = parts.length > 1 ? parts[parts.length - 1][0] : ""
   184→   184→  return (first + last).toUpperCase()
   185→   185→}

   187→   187→/**
   188→   188→ * Deals the people (and open roles) into columns round-robin, so neighbours
   189→   189→ * differ, then pads short columns with repeats until each holds `minPerColumn`
   190→   190→ * so a drifting loop never shows a gap. Repeats are flagged `dup` and are
   191→   191→ * hidden from assistive tech. Open roles are spread through the deck.
   192→   192→ */
   193→   193→export function dealColumns(count: number, roleCount: number, columns: number, minPerColumn: number) {
   194→   194→  const cols = Math.max(1, Math.floor(columns) || 1)
   195→   195→  const deck: { kind: "member" | "role"; index: number }[] = []
   196→   196→  for (let i = 0; i < count; i++) deck.push({ kind: "member", index: i })
   197→   197→  for (let r = 0; r < roleCount; r++) {
   198→   198→    const at = Math.min(deck.length, Math.round(((r + 1) * (count + roleCount)) / (roleCount + 1)) - 1 + r)
   199→   199→    deck.splice(Math.max(0, at), 0, { kind: "role", index: r })
   200→   200→  }
   201→   201→  const out: { kind: "member" | "role"; index: number; dup: boolean }[][] = []
   202→   202→  for (let c = 0; c < cols; c++) out.push([])
   203→   203→  deck.forEach((d, i) => out[i % cols].push({ kind: d.kind, index: d.index, dup: false }))
   204→   204→  if (!deck.length) return out
   205→   205→  // Pad from the far end of the column furthest away, so a repeat never sits
   206→   206→  // level with the same face in the next column.
   207→   207→  for (let c = 0; c < cols; c++) {
   208→   208→    let from = (c + Math.floor(cols / 2) + (cols % 2 ? 1 : 0)) % cols
   209→   209→    if (cols === 1) from = 0
   210→   210→    const pool = out[from].filter((e) => !e.dup).slice().reverse()
   211→   211→    const rest = deck.map((d) => ({ kind: d.kind, index: d.index, dup: false })).reverse()
   212→   212→    const candidates = pool.concat(rest)
   213→   213→    let k = 0
   214→   214→    let guard = 0
   215→   215→    while (out[c].length < minPerColumn && guard < candidates.length * 3) {
   216→   216→      const d = candidates[k % candidates.length]
   217→   217→      k++
   218→   218→      guard++
   219→   219→      const clash = out[c].some((e) => e.kind === d.kind && e.index === d.index)
   220→   220→      if (clash && guard < candidates.length) continue
   221→   221→      out[c].push({ kind: d.kind, index: d.index, dup: true })
   222→   222→    }
   223→   223→  }
   224→   224→  return out
   225→   225→}

   227→   227→/** Font size that makes the widest headline line fill `available` px. */
   228→   228→export function fitTitleSize(available: number, widestAt100: number, stretch: number, min: number, max: number) {
   229→   229→  if (!(available > 0) || !(widestAt100 > 0)) return min
   230→   230→  const size = (available / (widestAt100 * (stretch > 0 ? stretch : 1))) * 100
   231→   231→  return Math.max(min, Math.min(max, Math.floor(size * 10) / 10))
   232→   232→}

   235→   235→export function layoutFor(width: number, columns: number) {

   237→   237→  const wanted = Math.max(2, Math.min(4, Math.round(columns) || 3))
   238→   238→  if (width > 0 && width < 1000) return 2
   239→   239→  return wanted

   375→   375→/** Nudges a drifting column (or the phone strip) so a focused card is fully visible. */
   376→   376→function bringIntoView(card: HTMLElement) {
   377→   377→  const frame = card.closest(".dtr-wall, .dtr-strip") as HTMLElement | null
   378→   378→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   379→   379→  if (!frame || !track) return
   380→   380→  const f = frame.getBoundingClientRect()
   381→   381→  const r = card.getBoundingClientRect()
   382→   382→  const vertical = track.classList.contains("dtr-track")
   383→   383→  // Start from where the track is right now (it may still be easing back).
   384→   384→  const now = getComputedStyle(track).translate
   385→   385→  const parts = !now || now === "none" ? ["0", "0"] : now.split(" ")
   386→   386→  let x = parseFloat(parts[0]) || 0
   387→   387→  let y = parseFloat(parts[1] || "0") || 0
   388→   388→  const m = 24
   389→   389→  if (vertical) {
   390→   390→    if (r.top < f.top + m) y += f.top + m - r.top
   391→   391→    else if (r.bottom > f.bottom - m) y -= r.bottom - (f.bottom - m)
   392→   392→  } else {
   393→   393→    if (r.left < f.left + m) x += f.left + m - r.left
   394→   394→    else if (r.right > f.right - m) x -= r.right - (f.right - m)
   395→   395→  }
   396→   396→  track.style.translate = Math.round(x) + "px " + Math.round(y) + "px"
   397→   397→}

   609→   609→  // The shadow end is lifted toward the field so dark skin and dark clothes keep their detail.
   610→   610→  const mid = (palette && palette.mid) || mixHex(pal.hot, pal.highlight, 0.35)
   611→   611→  const tables = tritoneTables(mixHex(pal.shadow, pal.deep, 0.4), mid, mixHex(pal.highlight, pal.ink, 0.5))
   612→   612→  const still = !drift || !(driftSeconds > 0)
   613→   613→  const copies = still ? [0] : [0, 1]
   614→   614→  const [minPer, setMinPer] = React.useState(4)
   615→   615→  const plan = React.useMemo(
   616→   616→    () => dealColumns(members.length, openRoles.length, cols || 1, cols && !still ? minPer : 0),
   617→   617→    [members.length, openRoles.length, cols, minPer, still],
   618→   618→  )
   619→   619→
   620→   620→  const slotFor = (e: { kind: string; index: number; dup: boolean }, prefix: string): Slot =>
   621→   621→    e.kind === "role"
   622→   622→      ? { key: prefix, kind: "role", role: openRoles[e.index], dup: e.dup }
   623→   623→      : { key: prefix, kind: "member", member: members[e.index], dup: e.dup }
   624→   624→
   625→   625→  // Measure the container: choose the layout and fit the headline to its column.
   626→   626→  useIsoLayoutEffect(() => {
   627→   627→    const root = rootRef.current
   628→   628→    const copy = copyRef.current
   629→   629→    const head = titleRef.current
   630→   630→    if (!root || !copy || !head) return
   631→   631→    const fit = () => {
   632→   632→      const w = root.getBoundingClientRect().width
   633→   633→      const next = layoutFor(w, columns)
   634→   634→      setCols((c) => (c === next ? c : next))
   635→   635→      const spans = head.querySelectorAll(".dtr-line > span")
   636→   636→      let widest = 0
   637→   637→      const cur = parseFloat(getComputedStyle(head).fontSize) || 100
   638→   638→      spans.forEach((s) => {
   639→   639→        widest = Math.max(widest, (s as HTMLElement).offsetWidth)
   640→   640→      })
   641→   641→      const widestAt100 = (widest / cur) * 100
   642→   642→      const cs = getComputedStyle(copy)
   643→   643→      const avail = copy.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight)
   644→   644→      const h = root.getBoundingClientRect().height
   645→   645→      const strip = next === 0
   646→   646→      // Enough portraits per column that one loop is taller than the wall.
   647→   647→      if (next > 0) {
   648→   648→        const wallW = root.getBoundingClientRect().width * (next === 2 ? 0.5 : 0.58)
   649→   649→        const cardH = ((wallW - 14 * (next + 1)) / next) * (4 / 3) + 14
   650→   650→        const need = Math.max(3, Math.ceil(h / Math.max(80, cardH)))
   651→   651→        setMinPer((m) => (m === need ? m : need))
   652→   652→      }
   653→   653→      const maxByHeight = strip ? 200 : Math.max(56, (h * 0.52) / Math.max(1, lines.length + 0.45))
   654→   654→      // On the wall layout the widest line runs on into the first column of portraits, like a poster.
   655→   655→      const overhang = strip ? 1 : 1.18
   656→   656→      setTitleSize(fitTitleSize(avail * overhang, widestAt100, stretch, 40, Math.min(280, maxByHeight)))
   657→   657→    }
   658→   658→    fit()
   659→   659→    const ro = new ResizeObserver(fit)
   660→   660→    ro.observe(root)
   661→   661→    return () => ro.disconnect()
   662→   662→  }, [columns, stretch, title, fontFamily, uppercase, lines.length])
   663→   663→
   664→   664→  // Stop drifting while nobody can see it.
   665→   665→  React.useEffect(() => {
   666→   666→    const root = rootRef.current
   667→   667→    if (!root || typeof IntersectionObserver === "undefined") return
   668→   668→    const io = new IntersectionObserver(([entry]) => setOutOfView(!entry.isIntersecting), { threshold: 0 })
   669→   669→    io.observe(root)
   670→   670→    const vis = () => setTabHidden(document.hidden)
   671→   671→    document.addEventListener("visibilitychange", vis)
   672→   672→    return () => {
   673→   673→      io.disconnect()
   674→   674→      document.removeEventListener("visibilitychange", vis)
   675→   675→    }
   676→   676→  }, [])
   677→   677→
   678→   678→  const onEnter = React.useCallback((key: string) => setActive(key), [setActive])
   679→   679→  const onLeave = React.useCallback((key: string) => setActive((a) => (a === key ? null : a)), [setActive])
   680→   680→  const onToggle = React.useCallback((key: string) => setActive((a) => (a === key ? null : key)), [setActive])
   681→   681→
   682→   682→  // Tap outside an unfolded card folds it again.
   683→   683→  React.useEffect(() => {
   684→   684→    if (!active) return
   685→   685→    const down = (e: PointerEvent) => {
   686→   686→      if (e.pointerType === "mouse") return
   687→   687→      const t = e.target as HTMLElement | null
   688→   688→      if (!t || !t.closest || !t.closest(".dtr-card")) setActive(null)
   689→   689→    }
   690→   690→    document.addEventListener("pointerdown", down)
   691→   691→    return () => document.removeEventListener("pointerdown", down)
   692→   692→  }, [active])
   693→   693→
   694→   694→  const rootStyle = {
   695→   695→    ["--dtr-ground"]: pal.ground,
   696→   696→    ["--dtr-deep"]: pal.deep,
   697→   697→    ["--dtr-hot"]: pal.hot,
   698→   698→    ["--dtr-glow"]: pal.glow,
   699→   699→    ["--dtr-ink"]: pal.ink,
   700→   700→    ["--dtr-display"]: fontFamily,
   701→   701→    ["--dtr-stretch"]: String(stretch),
   702→   702→    ["--dtr-case"]: uppercase ? "uppercase" : "none",
   703→   703→    ["--dtr-cols"]: String(cols || 1),
   704→   704→    ["--dtr-ts"]: titleSize ? titleSize + "px" : undefined,
   705→   705→    ["--dtr-sdur"]: Math.max(8, (members.length + openRoles.length) * Math.max(1, driftSeconds) * 0.55) + "s",
   706→   706→    ["--dtr-h"]: height,
   707→   707→    height: cols === 0 || still ? "auto" : height,
   708→   708→    minHeight: cols === 0 ? undefined : still ? height : "600px",
   709→   709→  } as React.CSSProperties
   710→   710→
   711→   711→  const pauseButton = (
   712→   712→    <button
   713→   713→      type="button"
   714→   714→      className="dtr-pause"
   715→   715→      aria-pressed={paused}
   716→   716→      onClick={() => setPaused((p) => !p)}
   717→   717→    >
   718→   718→      <i aria-hidden="true" />
   719→   719→      {paused ? text.play : text.pause}
   720→   720→    </button>
   721→   721→  )
   722→   722→
   723→   723→  const phases = [0, 0.38, 0.71, 0.18]
   724→   724→
   725→   725→  return (
   726→   726→    <section
   727→   727→      ref={rootRef}
   728→   728→      className={"dtr " + className}
   729→   729→      style={rootStyle}
   730→   730→      aria-labelledby={titleId}
   731→   731→      data-mode={cols === 0 ? "strip" : "columns"}
   732→   732→      data-cols={cols || undefined}
   733→   733→      data-active={active ? "" : undefined}
   734→   734→      data-active-col={active && cols !== 0 ? active.split("-")[0] : undefined}
   735→   735→      data-paused={paused ? "" : undefined}
   736→   736→      data-static={still ? "" : undefined}
   737→   737→      data-offscreen={offscreen ? "" : undefined}
   738→   738→    >
   739→   739→      <style>{css}</style>
   740→   740→      <svg className="dtr-defs" aria-hidden="true" focusable="false">
   741→   741→        <filter id={filterId} colorInterpolationFilters="sRGB">
   742→   742→          <feColorMatrix type="matrix" values="0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0 0 0 1 0" />
   743→   743→          <feComponentTransfer>
   744→   744→            <feFuncR type="table" tableValues="0 0.13 0.5 0.9 1" />
   745→   745→            <feFuncG type="table" tableValues="0 0.13 0.5 0.9 1" />
   746→   746→            <feFuncB type="table" tableValues="0 0.13 0.5 0.9 1" />
   747→   747→          </feComponentTransfer>
   748→   748→          <feComponentTransfer>
   749→   749→            <feFuncR type="table" tableValues={tables[0]} />
   750→   750→            <feFuncG type="table" tableValues={tables[1]} />
   751→   751→            <feFuncB type="table" tableValues={tables[2]} />
   752→   752→          </feComponentTransfer>
   753→   753→        </filter>
   754→   754→      </svg>
   755→   755→      <div className="dtr-field" aria-hidden="true" />
   756→   756→      {grain ? <div className="dtr-grain" aria-hidden="true" /> : null}
   757→   757→      {cols !== 0 && !still ? pauseButton : null}
   758→   758→
   759→   759→      <div className="dtr-in">
   760→   760→        <div className="dtr-copy" ref={copyRef}>
   761→   761→          <h2 className="dtr-title" id={titleId} ref={titleRef}>
   762→   762→            {eyebrow ? <span className="dtr-eyebrow">{eyebrow} </span> : null}
   763→   763→            {lines.map((l, i) => (
   764→   764→              <span className="dtr-line" key={i} style={{ ["--i"]: i } as React.CSSProperties}>
   765→   765→                <span>
   766→   766→                  <span>{l}</span>
   767→   767→                </span>
   768→   768→                {i < lines.length - 1 ? " " : null}
   769→   769→              </span>
   770→   770→            ))}
   771→   771→          </h2>
   772→   772→          {intro ? <p className="dtr-intro">{intro}</p> : null}
   773→   773→          {cta || secondaryCta ? (
   774→   774→            <div className="dtr-actions">
   775→   775→              {cta ? (
   776→   776→                <a className="dtr-cta" href={cta.href}>
   777→   777→                  <span className="dtr-cta-label">
   778→   778→                    <span>{cta.label}</span>
   779→   779→                    <span aria-hidden="true">{cta.label}</span>
   780→   780→                  </span>
   781→   781→                  <span className="dtr-cta-arrow" aria-hidden="true">
   782→   782→                    →
   783→   783→                  </span>
   784→   784→                </a>
   785→   785→              ) : null}
   786→   786→              {secondaryCta ? (
   787→   787→                <a className="dtr-link" href={secondaryCta.href}>
   788→   788→                  {secondaryCta.label}
   789→   789→                </a>
   790→   790→              ) : null}
   791→   791→              {cols === 0 && !still ? pauseButton : null}
   792→   792→            </div>
   793→   793→          ) : null}
   794→   794→        </div>
   795→   795→
   796→   796→        {cols === 0 ? (
   797→   797→          <div className="dtr-strip" role="group" aria-label={text.team}>
   798→   798→            <div className="dtr-strip-track">
   799→   799→              {copies.map((copy) =>
   800→   800→                plan[0].map((e, i) => {
   801→   801→                  const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, "s" + copy + "-" + i)
   802→   802→                  return (
   803→   803→                    <Portrait
   804→   804→                      key={slot.key}
   805→   805→                      slot={slot}
   806→   806→                      on={active === slot.key}
   807→   807→                      filterId={filterId}
   808→   808→                      labels={text}
   809→   809→                      onEnter={onEnter}
   810→   810→                      onLeave={onLeave}
   811→   811→                      onToggle={onToggle}
   812→   812→                      onSelect={onMemberSelect}
   813→   813→                      follows={clickFollowsLink}
   814→   814→                    />
   815→   815→                  )
   816→   816→                }),
   817→   817→              )}
   818→   818→            </div>
   819→   819→          </div>
   820→   820→        ) : (
   821→   821→          <div className="dtr-wall" role="group" aria-label={text.team}>
   822→   822→            {plan.map((col, c) => {
   823→   823→              const dur = Math.max(6, col.length * Math.max(1, driftSeconds))
   824→   824→              const phase = phases[c % phases.length]

   826→   826→                <div
   827→   827→                  className="dtr-col"
   828→   828→                  key={c}
   829→   829→                  data-dir={c % 2 ? "down" : "up"}
   830→   830→                  style={{
   831→   831→                    ["--ci"]: c,
   832→   832→                    ["--dtr-dur"]: dur + "s",
   833→   833→                    ["--dtr-phase"]: -(dur * phase) + "s",
   834→   834→                    ["--dtr-rest"]: c % 2 ? 1 - phase : phase,
   835→   835→                  } as React.CSSProperties}
   836→   836→                >
   837→   837→                  <div className="dtr-track">
   838→   838→                    {copies.map((copy) =>
   839→   839→                      col.map((e, i) => {
   840→   840→                        const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, c + "-" + copy + "-" + i)
   841→   841→                        return (
   842→   842→                          <div key={slot.key}>
   843→   843→                            <Portrait
   844→   844→                              slot={slot}
   845→   845→                              on={active === slot.key}
   846→   846→                              filterId={filterId}
   847→   847→                              labels={text}
   848→   848→                              onEnter={onEnter}
   849→   849→                              onLeave={onLeave}
   850→   850→                              onToggle={onToggle}
   851→   851→                              onSelect={onMemberSelect}
   852→   852→                              follows={clickFollowsLink}
   853→   853→                            />
   854→   854→                          </div>
   855→   855→                        )
   856→   856→                      }),
   857→   857→                    )}
   858→   858→                  </div>
   859→   859→                </div>
   860→   860→              )
   861→   861→            })}
   862→   862→          </div>
   863→   863→        )}
   864→   864→      </div>
   865→   865→    </section>
   866→   866→  )
   867→   867→}
