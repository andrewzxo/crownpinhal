   103→function SubscriptionStatusBadge({ status }: { status: SubscriptionDTO["status"] }) {
   104→  if (status === "ACTIVE") {
   105→    return (
   106→      <Badge className="border border-emerald-500/40 bg-emerald-500/10 text-emerald-400">
   107→        Ativa
   108→      </Badge>
   109→    );
   110→  }
   111→  if (status === "CANCELLED") {
   112→    return <Badge variant="destructive">Cancelada</Badge>;
   113→  }
   114→  return (
   115→    <Badge className="border border-[#d9a441]/40 bg-[#d9a441]/10 text-[#d9a441]">Pendente</Badge>
   116→  );
   117→}
   118→
   119→function RoleBadge({ role }: { role: Role }) {
   120→  if (role === "OWNER") {
   121→    return (
   122→      <Badge className="border border-[#d9a441]/40 bg-[#d9a441]/10 text-[#d9a441]">OWNER</Badge>
   123→    );
   124→  }
   125→  if (role === "ADMIN") {
   126→    return (
   127→      <Badge variant="outline" className="border-[#3a3a42] text-zinc-400">
   128→        ADMIN
   129→      </Badge>
   130→    );
   131→  }
   132→  return <Badge variant="secondary" className="text-muted-foreground">CLIENT</Badge>;
   133→}

   159→
   160→function OverviewTab({ stats, statsLoading }: OverviewTabProps) {
   161→  const [recent, setRecent] = useState<SubscriptionDTO[] | null>(null);
   162→
   163→  useEffect(() => {
   164→    let cancelled = false;
   165→    void api<{ subscriptions: SubscriptionDTO[] }>("/api/owner/subscriptions")
   166→      .then((data) => {
   167→        if (cancelled) return;
   168→        const sorted = [...data.subscriptions].sort(
   169→          (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   170→        );
   171→        setRecent(sorted.slice(0, 5));
   172→      })
   173→      .catch((err) => {
   174→        if (cancelled) return;
   175→        toast.error(errMessage(err, "Erro ao carregar últimos pedidos."));
   176→        setRecent([]);
   177→      });
   178→    return () => {
   179→      cancelled = true;
   180→    };
   181→  }, []);
   182→
   183→  return (
   184→    <div className="space-y-6">
   185→      {/* KPIs */}
   186→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 xl:grid-cols-6">
   187→        {KPIS.map((kpi) => {
   188→          const Icon = kpi.icon;
   189→          const value = stats ? stats[kpi.key] : null;
   190→          const isGold = Boolean(kpi.gold);
   191→          const highlighted = isGold && (stats?.pendingSubscriptions ?? 0) > 0;
   192→          return (
   193→            <Card
   194→              key={kpi.key}
   195→              className={cn(
   196→                "gap-2 border-[#2b2b31] bg-card py-4",
   197→                highlighted && "border-[#d9a441]/50"
   198→              )}
   199→            >
   200→              <CardContent className="px-4">
   201→                <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   202→                  <Icon
   203→                    className={cn("size-4 shrink-0", isGold && "text-[#d9a441]")}
   204→                    aria-hidden
   205→                  />
   206→                  <span className="truncate">{kpi.label}</span>
   207→                </p>
   208→                {statsLoading || value == null ? (
   209→                  <Skeleton className="mt-2 h-9 w-16 bg-[#1a1a20]" />
   210→                ) : (
   211→                  <p
   212→                    className={cn(
   213→                      "font-display mt-1 text-4xl leading-none",
   214→                      isGold && "text-[#d9a441]"
   215→                    )}
   216→                  >
   217→                    {formatNumberBR(value)}
   218→                  </p>
   219→                )}
   220→              </CardContent>
   221→            </Card>
   222→          );
   223→        })}
   224→      </div>
   225→
   226→      {/* Últimos pedidos */}
   227→      <Card className="gap-4 border-[#2b2b31] bg-card py-4">
   228→        <CardHeader className="px-4 [.border-b]:pb-0">
   229→          <CardTitle className="font-display text-lg uppercase tracking-wide">
   230→            Últimos Pedidos
   231→          </CardTitle>
   232→          <CardDescription>Os 5 pedidos de assinatura mais recentes.</CardDescription>
   233→        </CardHeader>
   234→        <CardContent className="px-4">
   235→          {recent === null ? (
   236→            <TableSkeleton rows={5} height="h-14" />
   237→          ) : recent.length === 0 ? (
   238→            <EmptyState
   239→              icon={ShoppingBag}
   240→              title="Nenhum pedido ainda"
   241→              description="Os pedidos de assinatura aparecerão aqui."
   242→            />
   243→          ) : (
   244→            <ul className="divide-y divide-[#101013]">
   245→              {recent.map((s) => (
   246→                <li
   247→                  key={s.id}
   248→                  className="flex flex-wrap items-center justify-between gap-2 py-3"
   249→                >
   250→                  <div className="min-w-0">
   251→                    <p className="truncate text-sm font-medium">
   252→                      {s.user?.name ?? "Cliente"}
   253→                    </p>
   254→                    <p className="truncate text-xs text-muted-foreground">
   255→                      {s.plan.name} • {formatDateTimeBR(s.createdAt)}
   256→                    </p>
   257→                  </div>
   258→                  <SubscriptionStatusBadge status={s.status} />
   259→                </li>
   260→              ))}
   261→            </ul>
   262→          )}
   263→        </CardContent>
   264→      </Card>
   265→    </div>
   266→  );
   267→}

   273→function statusActionLabel(status: SubscriptionDTO["status"]): string {
   274→  if (status === "ACTIVE") return "Pedido ativado.";
   275→  if (status === "PENDING") return "Pedido marcado como pendente.";
   276→  return "Pedido cancelado.";

   281→  const [busyId, setBusyId] = useState<string | null>(null);
   282→  const [confirmDelete, setConfirmDelete] = useState<SubscriptionDTO | null>(null);
   283→  const [deleting, setDeleting] = useState(false);
   284→
   285→  const load = useCallback(async () => {
   286→    try {
   287→      const data = await api<{ subscriptions: SubscriptionDTO[] }>("/api/owner/subscriptions");
   288→      const sorted = [...data.subscriptions].sort(
   289→        (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   290→      );
   291→      setItems(sorted);
   292→    } catch (err) {
   293→      toast.error(errMessage(err, "Erro ao carregar pedidos."));
   294→      setItems([]);
   295→    }

   298→  useEffect(() => {
   299→    void load();

   303→    setBusyId(id);
   304→    try {
   305→      await api(`/api/owner/subscriptions/${id}`, {
   306→        method: "PUT",
   307→        body: JSON.stringify({ status }),
   308→      });
   309→      toast.success(statusActionLabel(status));
   310→      await Promise.all([load(), onChanged()]);
   311→    } catch (err) {
   312→      toast.error(errMessage(err, "Erro ao atualizar pedido."));
   313→    } finally {
   314→      setBusyId(null);
   315→    }
   316→  }

   321→      await api(`/api/owner/subscriptions/${id}`, { method: "DELETE" });

   325→      toast.error(errMessage(err, "Erro ao excluir pedido."));

   327→      setDeleting(false);

   487→function UsersTab() {

   495→  const [resetting, setResetting] = useState(false);
   496→
   497→  // Alterar papel
   498→  const [roleTarget, setRoleTarget] = useState<OwnerUserDTO | null>(null);
   499→  const [newRole, setNewRole] = useState<Role>("CLIENT");
   500→  const [changingRole, setChangingRole] = useState(false);
   501→
   502→  // Excluir
   503→  const [confirmDelete, setConfirmDelete] = useState<OwnerUserDTO | null>(null);
   504→  const [deleting, setDeleting] = useState(false);
   505→
   506→  const load = useCallback(async () => {
   507→    try {
   508→      const data = await api<{ users: OwnerUserDTO[] }>("/api/owner/users");
   509→      setUsers(data.users);
   510→    } catch (err) {
   511→      toast.error(errMessage(err, "Erro ao carregar usuários."));
   512→      setUsers([]);
   513→    }
   514→  }, []);
   515→
   516→  useEffect(() => {
   517→    void load();
   518→  }, [load]);
   519→
   520→  const filtered = (users ?? []).filter((u) => {
   521→    const q = query.trim().toLowerCase();
   522→    if (!q) return true;
   523→    return (
   524→      u.name.toLowerCase().includes(q) || u.email.toLowerCase().includes(q)
   525→    );
   526→  });
   527→
   528→  function openReset(user: OwnerUserDTO) {
   529→    setResetTarget(user);
   530→    setNewPassword("");
   531→  }
   532→
   533→  async function submitReset() {
   534→    if (!resetTarget) return;
   535→    if (newPassword.trim().length < 4) {
   536→      toast.error("A nova senha deve ter pelo menos 4 caracteres.");
   537→      return;
   538→    }
   539→    setResetting(true);
   540→    try {
   541→      await api(`/api/owner/users/${resetTarget.id}`, {
   542→        method: "PUT",
   543→        body: JSON.stringify({ newPassword: newPassword.trim() }),
   544→      });
   545→      toast.success(`Senha de ${resetTarget.name} redefinida com sucesso.`);
   546→      setResetTarget(null);
   547→      await load();
   548→    } catch (err) {
   549→      toast.error(errMessage(err, "Erro ao redefinir senha."));
   550→    } finally {
   551→      setResetting(false);
   552→    }
   553→  }
   554→
   555→  function openRole(user: OwnerUserDTO) {
   556→    setRoleTarget(user);
   557→    setNewRole(user.role);
   558→  }
   559→
   560→  async function submitRole() {
   561→    if (!roleTarget) return;
   562→    setChangingRole(true);
   563→    try {
   564→      await api(`/api/owner/users/${roleTarget.id}`, {
   565→        method: "PUT",
   566→        body: JSON.stringify({ role: newRole }),
   567→      });
   568→      toast.success(`Papel de ${roleTarget.name} alterado para ${newRole}.`);
   569→      setRoleTarget(null);
   570→      await load();
   571→    } catch (err) {
   572→      toast.error(errMessage(err, "Erro ao alterar papel."));
   573→    } finally {
   574→      setChangingRole(false);
   575→    }
   576→  }
   577→
   578→  async function removeUser(id: string) {
   579→    setDeleting(true);
   580→    try {
   581→      await api(`/api/owner/users/${id}`, { method: "DELETE" });
   582→      toast.success("Usuário excluído.");
   583→      setConfirmDelete(null);
   584→      await load();
   585→    } catch (err) {
