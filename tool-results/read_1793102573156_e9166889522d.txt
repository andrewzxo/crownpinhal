   160→   160→function OverviewTab({ stats, statsLoading }: OverviewTabProps) {
   161→   161→  const [recent, setRecent] = useState<SubscriptionDTO[] | null>(null);
   162→   162→
   163→   163→  useEffect(() => {
   164→   164→    let cancelled = false;
   165→   165→    void api<{ subscriptions: SubscriptionDTO[] }>("/api/owner/subscriptions")
   166→   166→      .then((data) => {
   167→   167→        if (cancelled) return;
   168→   168→        const sorted = [...data.subscriptions].sort(
   169→   169→          (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   170→   170→        );
   171→   171→        setRecent(sorted.slice(0, 5));
   172→   172→      })
   173→   173→      .catch((err) => {
   174→   174→        if (cancelled) return;
   175→   175→        toast.error(errMessage(err, "Erro ao carregar últimos pedidos."));
   176→   176→        setRecent([]);
   177→   177→      });
   178→   178→    return () => {
   179→   179→      cancelled = true;
   180→   180→    };
   181→   181→  }, []);
   182→   182→
   183→   183→  return (
   184→   184→    <div className="space-y-6">
   185→   185→      {/* KPIs */}
   186→   186→      <div className="grid grid-cols-2 gap-4 sm:grid-cols-3 xl:grid-cols-6">
   187→   187→        {KPIS.map((kpi) => {
   188→   188→          const Icon = kpi.icon;
   189→   189→          const value = stats ? stats[kpi.key] : null;
   190→   190→          const isGold = Boolean(kpi.gold);
   191→   191→          const highlighted = isGold && (stats?.pendingSubscriptions ?? 0) > 0;
   192→   192→          return (
   193→   193→            <Card
   194→   194→              key={kpi.key}
   195→   195→              className={cn(
   196→   196→                "gap-2 border-[#2b2b31] bg-card py-4",
   197→   197→                highlighted && "border-[#d9a441]/50"
   198→   198→              )}
   199→   199→            >
   200→   200→              <CardContent className="px-4">
   201→   201→                <p className="flex items-center gap-1.5 text-xs text-muted-foreground">
   202→   202→                  <Icon
   203→   203→                    className={cn("size-4 shrink-0", isGold && "text-[#d9a441]")}
   204→   204→                    aria-hidden
   205→   205→                  />
   206→   206→                  <span className="truncate">{kpi.label}</span>
   207→   207→                </p>
   208→   208→                {statsLoading || value == null ? (
   209→   209→                  <Skeleton className="mt-2 h-9 w-16 bg-[#1a1a20]" />
   210→   210→                ) : (
   211→   211→                  <p
   212→   212→                    className={cn(
   213→   213→                      "font-display mt-1 text-4xl leading-none",
   214→   214→                      isGold && "text-[#d9a441]"
   215→   215→                    )}
   216→   216→                  >
   217→   217→                    {formatNumberBR(value)}
   218→   218→                  </p>
   219→   219→                )}
   220→   220→              </CardContent>
   221→   221→            </Card>
   222→   222→          );
   223→   223→        })}
   224→   224→      </div>
   225→   225→
   226→   226→      {/* Últimos pedidos */}
   227→   227→      <Card className="gap-4 border-[#2b2b31] bg-card py-4">
   228→   228→        <CardHeader className="px-4 [.border-b]:pb-0">
   229→   229→          <CardTitle className="font-display text-lg uppercase tracking-wide">
   230→   230→            Últimos Pedidos
   231→   231→          </CardTitle>
   232→   232→          <CardDescription>Os 5 pedidos de assinatura mais recentes.</CardDescription>
   233→   233→        </CardHeader>
   234→   234→        <CardContent className="px-4">
   235→   235→          {recent === null ? (
   236→   236→            <TableSkeleton rows={5} height="h-14" />
   237→   237→          ) : recent.length === 0 ? (
   238→   238→            <EmptyState
   239→   239→              icon={ShoppingBag}
   240→   240→              title="Nenhum pedido ainda"
   241→   241→              description="Os pedidos de assinatura aparecerão aqui."
   242→   242→            />
   243→   243→          ) : (
   244→   244→            <ul className="divide-y divide-[#101013]">
   245→   245→              {recent.map((s) => (
   246→   246→                <li
   247→   247→                  key={s.id}
   248→   248→                  className="flex flex-wrap items-center justify-between gap-2 py-3"
   249→   249→                >
   250→   250→                  <div className="min-w-0">
   251→   251→                    <p className="truncate text-sm font-medium">
   252→   252→                      {s.user?.name ?? "Cliente"}
   253→   253→                    </p>
   254→   254→                    <p className="truncate text-xs text-muted-foreground">
   255→   255→                      {s.plan.name} • {formatDateTimeBR(s.createdAt)}
   256→   256→                    </p>
   257→   257→                  </div>

   259→   259→                </li>

   263→   263→        </CardContent>

   273→   273→function statusActionLabel(status: SubscriptionDTO["status"]): string {
   274→   274→  if (status === "ACTIVE") return "Pedido ativado.";
   275→   275→  if (status === "PENDING") return "Pedido marcado como pendente.";

   279→   279→function OrdersTab({ onChanged }: { onChanged: () => void }) {
   280→   280→  const [items, setItems] = useState<SubscriptionDTO[] | null>(null);
   281→   281→  const [busyId, setBusyId] = useState<string | null>(null);
   282→   282→  const [confirmDelete, setConfirmDelete] = useState<SubscriptionDTO | null>(null);
   283→   283→  const [deleting, setDeleting] = useState(false);
   284→   284→
   285→   285→  const load = useCallback(async () => {
   286→   286→    try {
   287→   287→      const data = await api<{ subscriptions: SubscriptionDTO[] }>("/api/owner/subscriptions");
   288→   288→      const sorted = [...data.subscriptions].sort(
   289→   289→        (a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime()
   290→   290→      );
   291→   291→      setItems(sorted);
   292→   292→    } catch (err) {
   293→   293→      toast.error(errMessage(err, "Erro ao carregar pedidos."));
   294→   294→      setItems([]);
   295→   295→    }
   296→   296→  }, []);
   297→   297→
   298→   298→  useEffect(() => {
   299→   299→    void load();
   300→   300→  }, [load]);
   301→   301→
   302→   302→  async function setStatus(id: string, status: SubscriptionDTO["status"]) {
   303→   303→    setBusyId(id);
   304→   304→    try {
   305→   305→      await api(`/api/owner/subscriptions/${id}`, {
   306→   306→        method: "PUT",
   307→   307→        body: JSON.stringify({ status }),
   308→   308→      });
   309→   309→      toast.success(statusActionLabel(status));
   310→   310→      await Promise.all([load(), onChanged()]);
   311→   311→    } catch (err) {
   312→   312→      toast.error(errMessage(err, "Erro ao atualizar pedido."));
   313→   313→    } finally {
   314→   314→      setBusyId(null);
   315→   315→    }
   316→   316→  }
   317→   317→
   318→   318→  async function remove(id: string) {
   319→   319→    setDeleting(true);
   320→   320→    try {
   321→   321→      await api(`/api/owner/subscriptions/${id}`, { method: "DELETE" });
   322→   322→      toast.success("Pedido excluído.");
   323→   323→      await Promise.all([load(), onChanged()]);
   324→   324→    } catch (err) {
   325→   325→      toast.error(errMessage(err, "Erro ao excluir pedido."));
   326→   326→    } finally {
   327→   327→      setDeleting(false);
   328→   328→    }
   329→   329→  }
   330→   330→
   331→   331→  return (
   332→   332→    <Card className="gap-4 border-[#2b2b31] bg-card py-4">
   333→   333→      <CardHeader className="px-4 [.border-b]:pb-0">
   334→   334→        <CardTitle className="font-display text-lg uppercase tracking-wide">Pedidos</CardTitle>
   335→   335→        <CardDescription>
   336→   336→          Todos os pedidos de assinatura recebidos pela academia.
   337→   337→        </CardDescription>
   338→   338→      </CardHeader>
   339→   339→      <CardContent className="px-4">
   340→   340→        {items === null ? (
   341→   341→          <TableSkeleton rows={6} height="h-14" />
   342→   342→        ) : items.length === 0 ? (
   343→   343→          <EmptyState
   344→   344→            icon={ShoppingBag}
   345→   345→            title="Nenhum pedido encontrado"
   346→   346→            description="Quando um cliente assinar um plano, o pedido aparecerá aqui."
   347→   347→          />
   348→   348→        ) : (
   349→   349→          <div className="max-h-[60vh] overflow-auto rounded-md border border-[#2b2b31] [scrollbar-width:thin] [&::-webkit-scrollbar]:h-1.5 [&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-[#2b2b31]">
   350→   350→            <Table>
   351→   351→              <TableHeader>
   352→   352→                <TableRow className="border-[#2b2b31] hover:bg-transparent">
   353→   353→                  <TableHead className="text-[#d9a441]">Data</TableHead>
   354→   354→                  <TableHead className="text-[#d9a441]">Cliente</TableHead>
   355→   355→                  <TableHead className="text-[#d9a441]">Plano</TableHead>
   356→   356→                  <TableHead className="text-[#d9a441]">Status</TableHead>
   357→   357→                  <TableHead className="text-[#d9a441]">Observações</TableHead>
   358→   358→                  <TableHead className="text-right text-[#d9a441]">Ações</TableHead>
   359→   359→                </TableRow>
   360→   360→              </TableHeader>
   361→   361→              <TableBody>
   362→   362→                {items.map((s) => (
   363→   363→                  <TableRow key={s.id} className="border-[#2b2b31]">
   364→   364→                    <TableCell className="whitespace-nowrap text-sm text-muted-foreground">
   365→   365→                      {formatDateTimeBR(s.createdAt)}
   366→   366→                    </TableCell>
   367→   367→                    <TableCell>
   368→   368→                      <p className="whitespace-nowrap text-sm font-medium">
   369→   369→                        {s.user?.name ?? "—"}
   370→   370→                      </p>
   371→   371→                      <p className="text-xs text-muted-foreground">{s.user?.email}</p>
   372→   372→                      {s.user?.phone ? (
   373→   373→                        <p className="text-xs text-muted-foreground">{s.user.phone}</p>
   374→   374→                      ) : null}
   375→   375→                    </TableCell>
   376→   376→                    <TableCell className="whitespace-nowrap text-sm">

   451→   451→      >
   452→   452→        <AlertDialogContent className="border-[#2b2b31] bg-[#1a1a20]">
   453→   453→          <AlertDialogHeader>
   454→   454→            <AlertDialogTitle className="font-display uppercase tracking-wide">

   460→   460→            </AlertDialogDescription>
   461→   461→          </AlertDialogHeader>
   462→   462→          <AlertDialogFooter>
   463→   463→            <AlertDialogCancel>Cancelar</AlertDialogCancel>
   464→   464→            <AlertDialogAction

   472→   472→            >
