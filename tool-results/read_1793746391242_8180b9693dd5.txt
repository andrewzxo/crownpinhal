     5→/*
     6→ * Duotone Team Roster: a "meet the people" section. A heavy display headline
     7→ * sits bottom-left on a burning colour field; beside it, the team hangs in
     8→ * tall columns of portraits that drift slowly in opposite directions. Every
     9→ * photo is pushed through one tritone (shadow, midtone, highlight) built from
    10→ * the palette, so a roster shot on ten different phones reads as one set.
    11→ *
    12→ * The signature: point at, tab to or tap a portrait and it lifts out of the
    13→ * wall, colour bleeds back into it, the rest of the wall sinks into the field,
    14→ * and its small craft chip unfolds into a card with the name, role, one line
    15→ * and a link. The columns hold still while you read.

    17→ * Content, photos, open roles, colours (four presets or your own), column
    18→ * count, drift speed, type and height are all props. Below ~700px the columns
    19→ * become one drifting strip above the headline. Reduced motion keeps the
    20→ * columns staggered and still. A pause button stops the drift at any time.

    22→ * Integração CROWN (Task 29): usado na seção de modalidades — cada "retrato"

   150→export function resolvePalette(preset: string, overrides: Partial<RosterPalette> | undefined): RosterPalette {
   151→  const table: Record<string, RosterPalette> = PRESETS
   152→  const base = table[preset] || PRESETS.ember
   153→  const out: RosterPalette = Object.assign({}, base)
   154→  if (overrides) for (const k of Object.keys(base) as (keyof RosterPalette)[]) if (overrides[k]) out[k] = overrides[k]!
   155→  return out
   156→}

   158→/** "#f80" or "#ff8800" to [r, g, b] in 0..1, for the SVG tritone tables. */
   159→export function hexToUnit(hex: string) {
   160→  let h = String(hex || "").trim().replace("#", "")
   161→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
   162→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return [0, 0, 0]
   163→  return [0, 2, 4].map((i) => Math.round((parseInt(h.slice(i, i + 2), 16) / 255) * 1000) / 1000)
   164→}
   165→
   166→/** Mixes two hex colours, t = 0 gives a, t = 1 gives b. */
   167→export function mixHex(a: string, b: string, t: number) {
   168→  const x = hexToUnit(a)
   169→  const y = hexToUnit(b)
   170→  return "#" + [0, 1, 2].map((i) => Math.round((x[i] + (y[i] - x[i]) * t) * 255).toString(16).padStart(2, "0")).join("")
   171→}

   173→/** One feFuncR/G/B table per channel: shadow, midtone, highlight. */
   174→export function tritoneTables(shadow: string, mid: string, highlight: string) {
   175→  const s = hexToUnit(shadow)
   176→  const m = hexToUnit(mid)
   177→  const h = hexToUnit(highlight)
   178→  return [0, 1, 2].map((c) => s[c] + " " + m[c] + " " + h[c])
   179→}

   181→export function initials(name: string) {
   182→  const parts = String(name || "").trim().split(/\s+/).filter(Boolean)
   183→  if (!parts.length) return "?"
   184→  const first = parts[0][0]
   185→  const last = parts.length > 1 ? parts[parts.length - 1][0] : ""
   186→  return (first + last).toUpperCase()
   187→}

   189→/**
   190→ * Deals the people (and open roles) into columns round-robin, so neighbours
   191→ * differ, then pads short columns with repeats until each holds `minPerColumn`
   192→ * so a drifting loop never shows a gap. Repeats are flagged `dup` and are
   193→ * hidden from assistive tech. Open roles are spread through the deck.
   194→ */
   195→export function dealColumns(count: number, roleCount: number, columns: number, minPerColumn: number) {
   196→  const cols = Math.max(1, Math.floor(columns) || 1)
   197→  const deck: { kind: "member" | "role"; index: number }[] = []
   198→  for (let i = 0; i < count; i++) deck.push({ kind: "member", index: i })
   199→  for (let r = 0; r < roleCount; r++) {
   200→    const at = Math.min(deck.length, Math.round(((r + 1) * (count + roleCount)) / (roleCount + 1)) - 1 + r)
   201→    deck.splice(Math.max(0, at), 0, { kind: "role", index: r })
   202→  }
   203→  const out: { kind: "member" | "role"; index: number; dup: boolean }[][] = []
   204→  for (let c = 0; c < cols; c++) out.push([])
   205→  deck.forEach((d, i) => out[i % cols].push({ kind: d.kind, index: d.index, dup: false }))
   206→  if (!deck.length) return out
   207→  // Pad from the far end of the column furthest away, so a repeat never sits
   208→  // level with the same face in the next column.
   209→  for (let c = 0; c < cols; c++) {
   210→    let from = (c + Math.floor(cols / 2) + (cols % 2 ? 1 : 0)) % cols
   211→    if (cols === 1) from = 0
   212→    const pool = out[from].filter((e) => !e.dup).slice().reverse()
   213→    const rest = deck.map((d) => ({ kind: d.kind, index: d.index, dup: false })).reverse()
   214→    const candidates = pool.concat(rest)
   215→    let k = 0
   216→    let guard = 0
   217→    while (out[c].length < minPerColumn && guard < candidates.length * 3) {
   218→      const d = candidates[k % candidates.length]
   219→      k++
   220→      guard++
   221→      const clash = out[c].some((e) => e.kind === d.kind && e.index === d.index)
   222→      if (clash && guard < candidates.length) continue
   223→      out[c].push({ kind: d.kind, index: d.index, dup: true })
   224→    }
   225→  }
   226→  return out
   227→}

   229→/** Font size that makes the widest headline line fill `available` px. */
   230→export function fitTitleSize(available: number, widestAt100: number, stretch: number, min: number, max: number) {
   231→  if (!(available > 0) || !(widestAt100 > 0)) return min
   232→  const size = (available / (widestAt100 * (stretch > 0 ? stretch : 1))) * 100
   233→  return Math.max(min, Math.min(max, Math.floor(size * 10) / 10))
   234→}

   236→/** Layout for a container width: a strip on phones, 2 columns on tablets. */
   237→export function layoutFor(width: number, columns: number) {
   238→  if (width > 0 && width < 700) return 0
   239→  const wanted = Math.max(2, Math.min(4, Math.round(columns) || 3))
   240→  if (width > 0 && width < 1000) return 2
   241→  return wanted
   242→}

   377→/** Nudges a drifting column (or the phone strip) so a focused card is fully visible. */
   378→function bringIntoView(card: HTMLElement) {
   379→  const frame = card.closest(".dtr-wall, .dtr-strip") as HTMLElement | null
   380→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   381→  if (!frame || !track) return
   382→  const f = frame.getBoundingClientRect()
   383→  const r = card.getBoundingClientRect()
   384→  const vertical = track.classList.contains("dtr-track")
   385→  // Start from where the track is right now (it may still be easing back).
   386→  const now = getComputedStyle(track).translate
   387→  const parts = !now || now === "none" ? ["0", "0"] : now.split(" ")
   388→  let x = parseFloat(parts[0]) || 0
   389→  let y = parseFloat(parts[1] || "0") || 0
   390→  const m = 24
   391→  if (vertical) {
   392→    if (r.top < f.top + m) y += f.top + m - r.top
   393→    else if (r.bottom > f.bottom - m) y -= r.bottom - (f.bottom - m)
   394→  } else {
   395→    if (r.left < f.left + m) x += f.left + m - r.left
   396→    else if (r.right > f.right - m) x -= r.right - (f.right - m)
   397→  }
   398→  track.style.translate = Math.round(x) + "px " + Math.round(y) + "px"
   399→}

   403→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   404→  if (track) track.style.translate = ""

   407→function Portrait(props: {
   408→  slot: Slot
   409→  on: boolean
   410→  filterId: string
   411→  labels: RosterLabels
   412→  onEnter: (key: string) => void
   413→  onLeave: (key: string) => void
   414→  onToggle: (key: string) => void
   415→  onSelect?: (member: TeamMember) => void
   416→  follows: boolean
   417→}) {
   418→  const { slot, on, filterId, labels, onEnter, onLeave, onToggle, onSelect, follows } = props
   419→  const [photoSrc, setPhotoSrc] = React.useState<string | null>(null)
   420→  const pointer = React.useRef("mouse")
   421→  const panelId = "p" + React.useId().replace(/[^a-zA-Z0-9]/g, "")
   422→  const hidden = slot.dup

   425→  const role = slot.kind === "role" ? slot.role : null
   426→  const craft = member ? member.craft || member.role : role ? role.craft || labels.hiring : ""
   427→  const label = member ? member.name + ", " + member.role : role ? labels.openRole + ": " + role.title : ""

   431→  const photo = photoSrc === null ? basePhoto : photoSrc
   432→  const onPhotoError = () =>
   433→    setPhotoSrc((cur) => {
   434→      const now = cur === null ? basePhoto : cur
   435→      return photoAlt && now !== photoAlt ? photoAlt : ""
   436→    })

   440→    if (!col) return
   441→    if (hold) col.setAttribute("data-hold", "")
   442→    else col.removeAttribute("data-hold")
   443→  }
   444→  const href = member ? member.href : role ? role.href : undefined
   445→  const linkLabel = member ? member.linkLabel || labels.memberLink : (role && role.linkLabel) || labels.roleLink
   446→  const exposure = member && member.exposure && member.exposure > 0 ? member.exposure : 1
   447→  const focus = (member && member.focus) || "50% 22%"
   448→  const followable = follows && Boolean(href) && !/^mailto:/i.test(String(href))
   449→  const clickable = Boolean((member && onSelect) || followable)
   450→
   451→  const follow = (card: HTMLElement) => {
   452→    const a = card.querySelector(".dtr-panel a") as HTMLAnchorElement | null
   453→    if (a) a.click()
   454→  }
   455→
   456→  return (
   457→    <div
   458→      className={"dtr-card" + (isRole ? " dtr-role" : "")}
   459→      data-on={on ? "" : undefined}
   460→      data-click={clickable ? "" : undefined}
   461→      aria-hidden={hidden ? true : undefined}
   462→      style={{ ["--dtr-cw" as string]: Math.min(26, craft.length) * 0.74 + 2.2 + "em" } as React.CSSProperties}
   463→      onPointerDown={(e) => {
   464→        pointer.current = e.pointerType
   465→      }}
   466→      onPointerEnter={(e) => {
   467→        if (e.pointerType !== "mouse") return
   468→        holdColumn(e.currentTarget, true)
   469→        // A card cut off by the frame slides fully into view (it stays under the pointer).
   470→        bringIntoView(e.currentTarget)
   471→        onEnter(slot.key)
   472→      }}
   473→      onPointerLeave={(e) => {
   474→        if (e.pointerType !== "mouse") return
   475→        holdColumn(e.currentTarget, false)
   476→        onLeave(slot.key)
   477→        if (!e.currentTarget.contains(document.activeElement)) releaseNudge(e.currentTarget)
   478→      }}
   479→      onBlur={(e) => {
   480→        if (e.currentTarget.contains(e.relatedTarget as Node | null)) return
   481→        holdColumn(e.currentTarget, false)
   482→        onLeave(slot.key)
   483→        releaseNudge(e.currentTarget)
   484→      }}
   485→    >
   486→      <div className="dtr-photo">
   487→        {photo ? (
   488→          <>
   489→            <img
   490→              className="dtr-duo"
   491→              src={photo}

   499→                filter: (exposure !== 1 ? "brightness(" + exposure + ") " : "") + "url(#" + filterId + ")",
   500→                objectPosition: focus,
   501→              }}
   502→              onError={onPhotoError}
   503→            />
   504→            <img
   505→              className="dtr-true"
   506→              src={photo}
   507→              alt=""
   508→              width={720}
   509→              height={960}
   510→              loading="lazy"
   511→              decoding="async"
   512→              draggable={false}
   513→              style={{ objectPosition: focus }}
   514→              onError={onPhotoError}
   515→            />
   516→          </>
   517→        ) : (
   518→          <div className="dtr-plate" aria-hidden="true">
   519→            <span>{member ? initials(member.name) : (role && role.mark) || "You?"}</span>
   520→            {role ? <em>{role.title}</em> : null}
   521→          </div>
   522→        )}
   523→      </div>
   524→      <span className="dtr-chip" aria-hidden="true">
   525→        {craft}
   526→      </span>
   527→      <button
   528→        type="button"
   529→        className="dtr-hit"
   530→        tabIndex={hidden ? -1 : 0}
   531→        aria-label={label}
   532→        aria-expanded={hidden ? undefined : on}
   533→        aria-controls={panelId}
   534→        onFocus={(e) => {
   535→          // Pointer focus is handled by hover and tap; keyboard focus unfolds here.
   536→          if (!e.currentTarget.matches(":focus-visible")) return
   537→          onEnter(slot.key)
   538→          bringIntoView(e.currentTarget.parentElement as HTMLElement)
   539→        }}
   540→        onClick={(e) => {
   541→          const card = e.currentTarget.parentElement as HTMLElement
   542→          if (member && onSelect) onSelect(member)
   543→          if (e.detail === 0) {
   544→            // Keyboard: Enter opens the card, a second Enter moves to its link.
   545→            if (!on) {
   546→              onEnter(slot.key)
   547→              bringIntoView(card)
   548→            } else {
   549→              const a = card.querySelector(".dtr-panel a") as HTMLElement | null
   550→              if (a) a.focus()
   551→            }
   552→            return
   553→          }
   554→          if (pointer.current === "mouse") {
   555→            if (followable) follow(card)
   556→            return
   557→          }
   558→          if (!on) bringIntoView(card)
   559→          onToggle(slot.key)
   560→        }}
   561→        onKeyDown={(e) => {
   562→          if (e.key === "Escape") onLeave(slot.key)
   563→        }}
   564→      />
   565→      <div
   566→        className="dtr-panel"
   567→        id={panelId}
   568→        aria-hidden={on ? undefined : true}
   569→        onKeyDown={(e) => {
   570→          if (e.key !== "Escape") return
   571→          const btn = e.currentTarget.parentElement?.querySelector(".dtr-hit") as HTMLElement | null
   572→          if (btn) btn.focus()
   573→          onLeave(slot.key)
   574→        }}
   575→      >
   576→        <p className="dtr-name">{member ? member.name : role ? role.title : ""}</p>
   577→        {member || (role && role.note) ? <p className="dtr-roleline">{member ? member.role : role ? role.note : ""}</p> : null}
   578→        {member && member.bio ? <p className="dtr-bio">{member.bio}</p> : null}
   579→        {href ? (
   580→          <a href={href} tabIndex={hidden || !on ? -1 : 0}>
   581→            {linkLabel} <span aria-hidden="true">↗</span>
   582→          </a>
   583→        ) : null}
   584→      </div>
   585→    </div>
   586→  )
   587→}

   697→  }, [])
   698→
   699→  const onEnter = React.useCallback((key: string) => setActive(key), [setActive])
   700→  const onLeave = React.useCallback((key: string) => setActive((a) => (a === key ? null : a)), [setActive])
   701→  const onToggle = React.useCallback((key: string) => setActive((a) => (a === key ? null : key)), [setActive])
   702→
   703→  // Tap outside an unfolded card folds it again.
   704→  React.useEffect(() => {
   705→    if (!active) return
   706→    const down = (e: PointerEvent) => {
   707→      if (e.pointerType === "mouse") return
   708→      const t = e.target as HTMLElement | null
   709→      if (!t || !t.closest || !t.closest(".dtr-card")) setActive(null)
   710→    }
   711→    document.addEventListener("pointerdown", down)
   712→    return () => document.removeEventListener("pointerdown", down)
   713→  }, [active])
   714→
   715→  const rootStyle = {
   716→    ["--dtr-ground"]: pal.ground,
   717→    ["--dtr-deep"]: pal.deep,
   718→    ["--dtr-hot"]: pal.hot,
   719→    ["--dtr-glow"]: pal.glow,
   720→    ["--dtr-ink"]: pal.ink,
   721→    ["--dtr-display"]: fontFamily,
   722→    ["--dtr-stretch"]: String(stretch),
   723→    ["--dtr-case"]: uppercase ? "uppercase" : "none",
   724→    ["--dtr-cols"]: String(cols || 1),
   725→    ["--dtr-ts"]: titleSize ? titleSize + "px" : undefined,
   726→    ["--dtr-sdur"]: Math.max(8, (members.length + openRoles.length) * Math.max(1, driftSeconds) * 0.55) + "s",
   727→    ["--dtr-h"]: height,
   728→    height: cols === 0 || still ? "auto" : height,
   729→    minHeight: cols === 0 ? undefined : still ? height : "600px",
   730→  } as React.CSSProperties
   731→
   732→  const pauseButton = (
   733→    <button
   734→      type="button"
   735→      className="dtr-pause"
   736→      aria-pressed={paused}
   737→      onClick={() => setPaused((p) => !p)}
   738→    >
   739→      <i aria-hidden="true" />
   740→      {paused ? text.play : text.pause}
   741→    </button>
   742→  )
   743→
   744→  const phases = [0, 0.38, 0.71, 0.18]
   745→
   746→  return (
   747→    <section
   748→      ref={rootRef}
   749→      className={"dtr " + className}
   750→      style={rootStyle}
   751→      aria-labelledby={titleId}
   752→      data-mode={cols === 0 ? "strip" : "columns"}
   753→      data-cols={cols || undefined}
   754→      data-active={active ? "" : undefined}
   755→      data-active-col={active && cols !== 0 ? active.split("-")[0] : undefined}
   756→      data-paused={paused ? "" : undefined}
   757→      data-static={still ? "" : undefined}
   758→      data-offscreen={offscreen ? "" : undefined}
   759→    >
   760→      <style>{css}</style>
   761→      <svg className="dtr-defs" aria-hidden="true" focusable="false">
   762→        <filter id={filterId} colorInterpolationFilters="sRGB">
   763→          <feColorMatrix type="matrix" values="0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0 0 0 1 0" />
   764→          <feComponentTransfer>
   765→            <feFuncR type="table" tableValues="0 0.13 0.5 0.9 1" />
   766→            <feFuncG type="table" tableValues="0 0.13 0.5 0.9 1" />
   767→            <feFuncB type="table" tableValues="0 0.13 0.5 0.9 1" />
   768→          </feComponentTransfer>
   769→          <feComponentTransfer>
   770→            <feFuncR type="table" tableValues={tables[0]} />
   771→            <feFuncG type="table" tableValues={tables[1]} />
   772→            <feFuncB type="table" tableValues={tables[2]} />
   773→          </feComponentTransfer>
   774→        </filter>
   775→      </svg>
   776→      <div className="dtr-field" aria-hidden="true" />
   777→      {grain ? <div className="dtr-grain" aria-hidden="true" /> : null}
   778→      {cols !== 0 && !still ? pauseButton : null}
   779→
   780→      <div className="dtr-in">
   781→        <div className="dtr-copy" ref={copyRef}>
   782→          <h2 className="dtr-title" id={titleId} ref={titleRef}>
   783→            {eyebrow ? <span className="dtr-eyebrow">{eyebrow} </span> : null}
   784→            {lines.map((l, i) => (
   785→              <span className="dtr-line" key={i} style={{ ["--i"]: i } as React.CSSProperties}>
   786→                <span>
   787→                  <span>{l}</span>
   788→                </span>
   789→                {i < lines.length - 1 ? " " : null}
   790→              </span>
   791→            ))}
   792→          </h2>
   793→          {intro ? <p className="dtr-intro">{intro}</p> : null}
   794→          {cta || secondaryCta ? (
   795→            <div className="dtr-actions">
   796→              {cta ? (
   797→                <a className="dtr-cta" href={cta.href}>
   798→                  <span className="dtr-cta-label">
   799→                    <span>{cta.label}</span>
   800→                    <span aria-hidden="true">{cta.label}</span>
   801→                  </span>
   802→                  <span className="dtr-cta-arrow" aria-hidden="true">
   803→                    →
   804→                  </span>
   805→                </a>
   806→              ) : null}
   807→              {secondaryCta ? (
   808→                <a className="dtr-link" href={secondaryCta.href}>
   809→                  {secondaryCta.label}
   810→                </a>
   811→              ) : null}
   812→              {cols === 0 && !still ? pauseButton : null}
   813→            </div>
   814→          ) : null}
   815→        </div>
   816→
   817→        {cols === 0 ? (
   818→          <div className="dtr-strip" role="group" aria-label={text.team}>
   819→            <div className="dtr-strip-track">
   820→              {copies.map((copy) =>
   821→                plan[0].map((e, i) => {
   822→                  const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, "s" + copy + "-" + i)
   823→                  return (
   824→                    <Portrait
   825→                      key={slot.key}
   826→                      slot={slot}
   827→                      on={active === slot.key}
   828→                      filterId={filterId}
   829→                      labels={text}
   830→                      onEnter={onEnter}
   831→                      onLeave={onLeave}
   832→                      onToggle={onToggle}
   833→                      onSelect={onMemberSelect}
   834→                      follows={clickFollowsLink}
   835→                    />
   836→                  )
   837→                }),
   838→              )}
   839→            </div>
   840→          </div>
   841→        ) : (
   842→          <div className="dtr-wall" role="group" aria-label={text.team}>
   843→            {plan.map((col, c) => {
   844→              const dur = Math.max(6, col.length * Math.max(1, driftSeconds))
   845→              const phase = phases[c % phases.length]
   846→              return (
   847→                <div
   848→                  className="dtr-col"
   849→                  key={c}
   850→                  data-dir={c % 2 ? "down" : "up"}
   851→                  style={{
   852→                    ["--ci"]: c,
   853→                    ["--dtr-dur"]: dur + "s",
   854→                    ["--dtr-phase"]: -(dur * phase) + "s",
   855→                    ["--dtr-rest"]: c % 2 ? 1 - phase : phase,
   856→                  } as React.CSSProperties}
   857→                >
   858→                  <div className="dtr-track">
   859→                    {copies.map((copy) =>
   860→                      col.map((e, i) => {
   861→                        const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, c + "-" + copy + "-" + i)
   862→                        return (
   863→                          <div key={slot.key}>
   864→                            <Portrait
   865→                              slot={slot}
   866→                              on={active === slot.key}
   867→                              filterId={filterId}
   868→                              labels={text}
   869→                              onEnter={onEnter}
   870→                              onLeave={onLeave}
   871→                              onToggle={onToggle}
   872→                              onSelect={onMemberSelect}
   873→                              follows={clickFollowsLink}
   874→                            />
   875→                          </div>
   876→                        )
   877→                      }),
   878→                    )}
   879→                  </div>
   880→                </div>
   881→              )
   882→            })}
   883→          </div>
   884→        )}
   885→      </div>
   886→    </section>
   887→  )
