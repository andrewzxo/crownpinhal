   168→/** "#f80" or "#ff8800" to [r, g, b] in 0..1, for the SVG tritone tables. */
   169→export function hexToUnit(hex: string) {
   170→  let h = String(hex || "").trim().replace("#", "")
   171→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
   172→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return [0, 0, 0]
   173→  return [0, 2, 4].map((i) => Math.round((parseInt(h.slice(i, i + 2), 16) / 255) * 1000) / 1000)
   174→}
   175→
   176→/** Mixes two hex colours, t = 0 gives a, t = 1 gives b. */
   177→export function mixHex(a: string, b: string, t: number) {
   178→  const x = hexToUnit(a)
   179→  const y = hexToUnit(b)
   180→  return "#" + [0, 1, 2].map((i) => Math.round((x[i] + (y[i] - x[i]) * t) * 255).toString(16).padStart(2, "0")).join("")
   181→}

   184→export function tritoneTables(shadow: string, mid: string, highlight: string) {

   191→export function initials(name: string) {
   192→  const parts = String(name || "").trim().split(/\s+/).filter(Boolean)
   193→  if (!parts.length) return "?"
   194→  const first = parts[0][0]
   195→  const last = parts.length > 1 ? parts[parts.length - 1][0] : ""
   196→  return (first + last).toUpperCase()

   199→/**
   200→ * Deals the people (and open roles) into columns round-robin, so neighbours
   201→ * differ, then pads short columns with repeats until each holds `minPerColumn`
   202→ * so a drifting loop never shows a gap. Repeats are flagged `dup` and are
   203→ * hidden from assistive tech. Open roles are spread through the deck.
   204→ */
   205→export function dealColumns(count: number, roleCount: number, columns: number, minPerColumn: number) {
   206→  const cols = Math.max(1, Math.floor(columns) || 1)
   207→  const deck: any[] = []
   208→  for (let i = 0; i < count; i++) deck.push({ kind: "member", index: i })
   209→  for (let r = 0; r < roleCount; r++) {
   210→    const at = Math.min(deck.length, Math.round(((r + 1) * (count + roleCount)) / (roleCount + 1)) - 1 + r)
   211→    deck.splice(Math.max(0, at), 0, { kind: "role", index: r })
   212→  }
   213→  const out: any[][] = []
   214→  for (let c = 0; c < cols; c++) out.push([])
   215→  deck.forEach((d, i) => out[i % cols].push({ kind: d.kind, index: d.index, dup: false }))
   216→  if (!deck.length) return out
   217→  // Pad from the far end of the column furthest away, so a repeat never sits
   218→  // level with the same face in the next column.
   219→  for (let c = 0; c < cols; c++) {
   220→    let from = (c + Math.floor(cols / 2) + (cols % 2 ? 1 : 0)) % cols
   221→    if (cols === 1) from = 0
   222→    const pool = out[from].filter((e) => !e.dup).slice().reverse()
   223→    const rest = deck.map((d) => ({ kind: d.kind, index: d.index, dup: false })).reverse()
   224→    const candidates = pool.concat(rest)
   225→    let k = 0
   226→    let guard = 0
   227→    while (out[c].length < minPerColumn && guard < candidates.length * 3) {
   228→      const d = candidates[k % candidates.length]
   229→      k++
   230→      guard++
   231→      const clash = out[c].some((e) => e.kind === d.kind && e.index === d.index)
   232→      if (clash && guard < candidates.length) continue
   233→      out[c].push({ kind: d.kind, index: d.index, dup: true })
   234→    }
   235→  }
   236→  return out
   237→}

   240→export function fitTitleSize(available: number, widestAt100: number, stretch: number, min: number, max: number) {
   241→  if (!(available > 0) || !(widestAt100 > 0)) return min
   242→  const size = (available / (widestAt100 * (stretch > 0 ? stretch : 1))) * 100
   243→  return Math.max(min, Math.min(max, Math.floor(size * 10) / 10))
   244→}

   404→/** Nudges a drifting column (or the phone strip) so a focused card is fully visible. */
   405→function bringIntoView(card: HTMLElement) {
   406→  const frame = card.closest(".dtr-wall, .dtr-strip") as HTMLElement | null
   407→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   408→  if (!frame || !track) return
   409→  const f = frame.getBoundingClientRect()
   410→  const r = card.getBoundingClientRect()
   411→  const vertical = track.classList.contains("dtr-track")
   412→  // Start from where the track is right now (it may still be easing back).
   413→  const now = getComputedStyle(track).translate
   414→  const parts = !now || now === "none" ? ["0", "0"] : now.split(" ")
   415→  let x = parseFloat(parts[0]) || 0
   416→  let y = parseFloat(parts[1] || "0") || 0
   417→  const m = 24
   418→  if (vertical) {
   419→    if (r.top < f.top + m) y += f.top + m - r.top
   420→    else if (r.bottom > f.bottom - m) y -= r.bottom - (f.bottom - m)
   421→  } else {
   422→    if (r.left < f.left + m) x += f.left + m - r.left
   423→    else if (r.right > f.right - m) x -= r.right - (f.right - m)
   424→  }
   425→  track.style.translate = Math.round(x) + "px " + Math.round(y) + "px"
   426→}

   428→/** Lets the track ease back into its loop once focus moves on. */
   429→function releaseNudge(card: HTMLElement) {
   430→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   431→  if (track) track.style.translate = ""

   591→        ) : null}
   592→      </div>
   593→    </div>

   667→      const cur = parseFloat(getComputedStyle(head).fontSize) || 100
   668→      spans.forEach((s) => {
   669→        widest = Math.max(widest, (s as HTMLElement).offsetWidth)
   670→      })
   671→      const widestAt100 = (widest / cur) * 100
   672→      const cs = getComputedStyle(copy)
   673→      const avail = copy.clientWidth - parseFloat(cs.paddingLeft) - parseFloat(cs.paddingRight)
   674→      const h = root.getBoundingClientRect().height
   675→      const strip = next === 0
   676→      // Enough portraits per column that one loop is taller than the wall.
   677→      if (next > 0) {
   678→        const wallW = root.getBoundingClientRect().width * (next === 2 ? 0.5 : 0.58)
   679→        const cardH = ((wallW - 14 * (next + 1)) / next) * (4 / 3) + 14
   680→        const need = Math.max(3, Math.ceil(h / Math.max(80, cardH)))
   681→        setMinPer((m) => (m === need ? m : need))
   682→      }
   683→      const maxByHeight = strip ? 200 : Math.max(56, (h * 0.52) / Math.max(1, lines.length + 0.45))
   684→      // On the wall layout the widest line runs on into the first column of portraits, like a poster.
   685→      const overhang = strip ? 1 : 1.18
   686→      setTitleSize(fitTitleSize(avail * overhang, widestAt100, stretch, 40, Math.min(280, maxByHeight)))
   687→    }
   688→    fit()
   689→    const ro = new ResizeObserver(fit)
   690→    ro.observe(root)
   691→    return () => ro.disconnect()
   692→  }, [columns, stretch, title, fontFamily, uppercase, lines.length])
   693→
   694→  // Stop drifting while nobody can see it.
   695→  React.useEffect(() => {
   696→    const root = rootRef.current
   697→    if (!root || typeof IntersectionObserver === "undefined") return
   698→    const io = new IntersectionObserver(([entry]) => setOutOfView(!entry.isIntersecting), { threshold: 0 })
   699→    io.observe(root)
   700→    const vis = () => setTabHidden(document.hidden)
   701→    document.addEventListener("visibilitychange", vis)
   702→    return () => {
   703→      io.disconnect()
   704→      document.removeEventListener("visibilitychange", vis)
   705→    }
   706→  }, [])
   707→
   708→  const onEnter = React.useCallback((key: string) => setActive(key), [])
   709→  const onLeave = React.useCallback((key: string) => setActive((a) => (a === key ? null : a)), [])
   710→  const onToggle = React.useCallback((key: string) => setActive((a) => (a === key ? null : key)), [])
   711→
   712→  // Tap outside an unfolded card folds it again.
   713→  React.useEffect(() => {
   714→    if (!active) return
   715→    const down = (e: PointerEvent) => {
   716→      if (e.pointerType === "mouse") return
   717→      const t = e.target as HTMLElement | null
   718→      if (!t || !t.closest || !t.closest(".dtr-card")) setActive(null)
   719→    }
   720→    document.addEventListener("pointerdown", down)
   721→    return () => document.removeEventListener("pointerdown", down)
   722→  }, [active])
   723→
   724→  const rootStyle = {
   725→    ["--dtr-ground" as any]: pal.ground,
   726→    ["--dtr-deep" as any]: pal.deep,
   727→    ["--dtr-hot" as any]: pal.hot,
   728→    ["--dtr-glow" as any]: pal.glow,
   729→    ["--dtr-ink" as any]: pal.ink,
   730→    ["--dtr-display" as any]: fontFamily,
   731→    ["--dtr-stretch" as any]: String(stretch),
   732→    ["--dtr-case" as any]: uppercase ? "uppercase" : "none",
   733→    ["--dtr-cols" as any]: String(cols || 1),
   734→    ["--dtr-ts" as any]: titleSize ? titleSize + "px" : undefined,
   735→    ["--dtr-sdur" as any]: Math.max(8, (members.length + openRoles.length) * Math.max(1, driftSeconds) * 0.55) + "s",
   736→    ["--dtr-h" as any]: height,
   737→    height: cols === 0 || still ? "auto" : height,
   738→    minHeight: cols === 0 ? undefined : still ? height : "600px",
   739→  } as React.CSSProperties
   740→
   741→  const pauseButton = (
   742→    <button
   743→      type="button"
   744→      className="dtr-pause"
   745→      aria-pressed={paused}
   746→      onClick={() => setPaused((p) => !p)}
   747→    >
   748→      <i aria-hidden="true" />
   749→      {paused ? text.play : text.pause}
   750→    </button>
   751→  )
   752→
   753→  const phases = [0, 0.38, 0.71, 0.18]
   754→
   755→  return (
   756→    <section
   757→      ref={rootRef}
   758→      className={"dtr " + className}
   759→      style={rootStyle}
   760→      aria-labelledby={titleId}
   761→      data-mode={cols === 0 ? "strip" : "columns"}
   762→      data-cols={cols || undefined}
   763→      data-active={active ? "" : undefined}
   764→      data-active-col={active && cols !== 0 ? active.split("-")[0] : undefined}
   765→      data-paused={paused ? "" : undefined}
   766→      data-static={still ? "" : undefined}
   767→      data-offscreen={offscreen ? "" : undefined}
   768→    >
   769→      <style>{css}</style>
   770→      <svg className="dtr-defs" aria-hidden="true" focusable="false">
   771→        <filter id={filterId} colorInterpolationFilters="sRGB">
   772→          <feColorMatrix type="matrix" values="0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0.2126 0.7152 0.0722 0 0  0 0 0 1 0" />
   773→          <feComponentTransfer>
   774→            <feFuncR type="table" tableValues="0 0.13 0.5 0.9 1" />
   775→            <feFuncG type="table" tableValues="0 0.13 0.5 0.9 1" />
   776→            <feFuncB type="table" tableValues="0 0.13 0.5 0.9 1" />
   777→          </feComponentTransfer>
   778→          <feComponentTransfer>
   779→            <feFuncR type="table" tableValues={tables[0]} />
   780→            <feFuncG type="table" tableValues={tables[1]} />
   781→            <feFuncB type="table" tableValues={tables[2]} />
   782→          </feComponentTransfer>
   783→        </filter>
   784→      </svg>
   785→      <div className="dtr-field" aria-hidden="true" />
   786→      {grain ? <div className="dtr-grain" aria-hidden="true" /> : null}
   787→      {cols !== 0 && !still ? pauseButton : null}
   788→
   789→      <div className="dtr-in">
   790→        <div className="dtr-copy" ref={copyRef}>
   791→          <h2 className="dtr-title" id={titleId} ref={titleRef}>
   792→            {eyebrow ? <span className="dtr-eyebrow">{eyebrow} </span> : null}
   793→            {lines.map((l, i) => (
   794→              <span className="dtr-line" key={i} style={{ ["--i" as any]: i }}>
   795→                <span>
   796→                  <span>{l}</span>
   797→                </span>
   798→                {i < lines.length - 1 ? " " : null}
   799→              </span>
   800→            ))}
   801→          </h2>
   802→          {intro ? <p className="dtr-intro">{intro}</p> : null}
   803→          {cta || secondaryCta ? (
   804→            <div className="dtr-actions">
   805→              {cta ? (
   806→                <a className="dtr-cta" href={cta.href}>
   807→                  <span className="dtr-cta-label">
   808→                    <span>{cta.label}</span>
   809→                    <span aria-hidden="true">{cta.label}</span>
   810→                  </span>
   811→                  <span className="dtr-cta-arrow" aria-hidden="true">
   812→                    →
   813→                  </span>
   814→                </a>
   815→              ) : null}
   816→              {secondaryCta ? (
   817→                <a className="dtr-link" href={secondaryCta.href}>
   818→                  {secondaryCta.label}
   819→                </a>
   820→              ) : null}
   821→              {cols === 0 && !still ? pauseButton : null}
   822→            </div>
   823→          ) : null}
   824→        </div>
   825→
   826→        {cols === 0 ? (
   827→          <div className="dtr-strip" role="group" aria-label={text.team}>
   828→            <div className="dtr-strip-track">
   829→              {copies.map((copy) =>
   830→                plan[0].map((e, i) => {
   831→                  const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, "s" + copy + "-" + i)
   832→                  return (
   833→                    <Portrait
   834→                      key={slot.key}
   835→                      slot={slot}
   836→                      on={active === slot.key}
   837→                      filterId={filterId}
   838→                      labels={text}
   839→                      onEnter={onEnter}
   840→                      onLeave={onLeave}
   841→                      onToggle={onToggle}
   842→                      onSelect={onMemberSelect}
   843→                      follows={clickFollowsLink}
   844→                    />
   845→                  )
   846→                }),
   847→              )}
   848→            </div>
   849→          </div>
   850→        ) : (
   851→          <div className="dtr-wall" role="group" aria-label={text.team}>
   852→            {plan.map((col, c) => {
   853→              const dur = Math.max(6, col.length * Math.max(1, driftSeconds))
   854→              const phase = phases[c % phases.length]
   855→              return (
   856→                <div
   857→                  className="dtr-col"
   858→                  key={c}
   859→                  data-dir={c % 2 ? "down" : "up"}
   860→                  style={{
   861→                    ["--ci" as any]: c,
   862→                    ["--dtr-dur" as any]: dur + "s",
   863→                    ["--dtr-phase" as any]: -(dur * phase) + "s",
   864→                    ["--dtr-rest" as any]: c % 2 ? 1 - phase : phase,
   865→                  }}
   866→                >
   867→                  <div className="dtr-track">
   868→                    {copies.map((copy) =>
   869→                      col.map((e, i) => {
   870→                        const slot = slotFor({ ...e, dup: e.dup || copy === 1 }, c + "-" + copy + "-" + i)
   871→                        return (
   872→                          <div key={slot.key}>
   873→                            <Portrait
   874→                              slot={slot}
   875→                              on={active === slot.key}
   876→                              filterId={filterId}
   877→                              labels={text}
   878→                              onEnter={onEnter}
   879→                              onLeave={onLeave}
   880→                              onToggle={onToggle}
   881→                              onSelect={onMemberSelect}
   882→                              follows={clickFollowsLink}
   883→                            />
   884→                          </div>
   885→                        )
   886→                      }),
   887→                    )}
   888→                  </div>
   889→                </div>
   890→              )
   891→            })}
   892→          </div>
   893→        )}
   894→      </div>
   895→    </section>
   896→  )
   897→}
