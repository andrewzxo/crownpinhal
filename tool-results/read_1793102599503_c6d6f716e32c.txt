   501→interface ScheduleForm {
   502→  dayOfWeek: number;
   503→  startTime: string;
   504→  endTime: string;
   505→  instructor: string;
   506→}
   507→
   508→function SchedulesTab() {
   509→  const refreshContent = useAppStore((s) => s.refreshContent);
   510→
   511→  const [modalities, setModalities] = useState<ModalityDTO[]>([]);
   512→  const [selectedId, setSelectedId] = useState<string>("");
   513→  const [loading, setLoading] = useState(true);
   514→  const [dialogOpen, setDialogOpen] = useState(false);
   515→  const [editing, setEditing] = useState<ScheduleDTO | null>(null);
   516→  const [saving, setSaving] = useState(false);
   517→  const [form, setForm] = useState<ScheduleForm>({
   518→    dayOfWeek: 1,
   519→    startTime: "07:00",
   520→    endTime: "08:00",
   521→    instructor: "",
   522→  });
   523→
   524→  const load = useCallback(async () => {
   525→    setLoading(true);
   526→    try {
   527→      const data = await api<{ modalities: ModalityDTO[] }>("/api/admin/modalities");
   528→      setModalities(data.modalities);
   529→      setSelectedId((prev) => prev || data.modalities[0]?.id || "");
   530→    } catch (err) {
   531→      toast.error(errMessage(err, "Erro ao carregar modalidades."));
   532→    } finally {
   533→      setLoading(false);
   534→    }
   535→  }, []);
   536→
   537→  useEffect(() => {
   538→    void load();
   539→  }, [load]);
   540→
   541→  const selected = modalities.find((m) => m.id === selectedId) ?? null;
   542→
   543→  const grouped = useMemo(() => {
   544→    const list = selected?.schedules ?? [];
   545→    return DAY_ORDER.map((day) => ({
   546→      day,
   547→      schedules: list
   548→        .filter((s) => s.dayOfWeek === day)
   549→        .sort((a, b) => a.startTime.localeCompare(b.startTime)),
   550→    })).filter((g) => g.schedules.length > 0);
   551→  }, [selected]);
   552→
   553→  function openCreate() {
   554→    setEditing(null);
   555→    setForm({ dayOfWeek: 1, startTime: "07:00", endTime: "08:00", instructor: "" });
   556→    setDialogOpen(true);
   557→  }
   558→
   559→  function openEdit(schedule: ScheduleDTO) {
   560→    setEditing(schedule);
   561→    setForm({
   562→      dayOfWeek: schedule.dayOfWeek,
   563→      startTime: formatTime(schedule.startTime),
   564→      endTime: formatTime(schedule.endTime),
   565→      instructor: schedule.instructor ?? "",
   566→    });
   567→    setDialogOpen(true);
   568→  }
   569→
   570→  async function save() {
   571→    if (!selected) return;
   572→    if (!form.startTime || !form.endTime) {
   573→      toast.error("Informe os horários de início e fim.");
   574→      return;
   575→    }
   576→    setSaving(true);
   577→    try {
   578→      const payload = {
   579→        modalityId: selected.id,
   580→        dayOfWeek: form.dayOfWeek,
   581→        startTime: form.startTime,
   582→        endTime: form.endTime,
   583→        instructor: form.instructor.trim() || null,
   584→      };
   585→      if (editing) {
   586→        await api(`/api/admin/schedules/${editing.id}`, {
   587→          method: "PUT",
   588→          body: JSON.stringify(payload),
   589→        });
   590→        toast.success("Horário atualizado com sucesso.");
   591→      } else {
   592→        await api("/api/admin/schedules", {
   593→          method: "POST",
   594→          body: JSON.stringify(payload),
   595→        });
   596→        toast.success("Horário criado com sucesso.");
   597→      }
   598→      setDialogOpen(false);
   599→      await Promise.all([load(), refreshContent()]);
   600→    } catch (err) {
   601→      toast.error(errMessage(err, "Erro ao salvar horário."));
   602→    } finally {
   603→      setSaving(false);
   604→    }
   605→  }
   606→
   607→  async function remove(id: string) {
   608→    try {
   609→      await api(`/api/admin/schedules/${id}`, { method: "DELETE" });
   610→      toast.success("Horário excluído.");
