   168→   168→/** "#f80" or "#ff8800" to [r, g, b] in 0..1, for the SVG tritone tables. */
   169→   169→export function hexToUnit(hex: string) {
   170→   170→  let h = String(hex || "").trim().replace("#", "")
   171→   171→  if (h.length === 3) h = h[0] + h[0] + h[1] + h[1] + h[2] + h[2]
   172→   172→  if (!/^[0-9a-fA-F]{6}$/.test(h)) return [0, 0, 0]
   173→   173→  return [0, 2, 4].map((i) => Math.round((parseInt(h.slice(i, i + 2), 16) / 255) * 1000) / 1000)
   174→   174→}
   175→   175→
   176→   176→/** Mixes two hex colours, t = 0 gives a, t = 1 gives b. */
   177→   177→export function mixHex(a: string, b: string, t: number) {
   178→   178→  const x = hexToUnit(a)
   179→   179→  const y = hexToUnit(b)
   180→   180→  return "#" + [0, 1, 2].map((i) => Math.round((x[i] + (y[i] - x[i]) * t) * 255).toString(16).padStart(2, "0")).join("")
   181→   181→}

   184→   184→export function tritoneTables(shadow: string, mid: string, highlight: string) {

   189→   189→}

   199→   199→/**
   200→   200→ * Deals the people (and open roles) into columns round-robin, so neighbours
   201→   201→ * differ, then pads short columns with repeats until each holds `minPerColumn`
   202→   202→ * so a drifting loop never shows a gap. Repeats are flagged `dup` and are
   203→   203→ * hidden from assistive tech. Open roles are spread through the deck.
   204→   204→ */
   205→   205→export function dealColumns(count: number, roleCount: number, columns: number, minPerColumn: number) {
   206→   206→  const cols = Math.max(1, Math.floor(columns) || 1)
   207→   207→  const deck: any[] = []
   208→   208→  for (let i = 0; i < count; i++) deck.push({ kind: "member", index: i })
   209→   209→  for (let r = 0; r < roleCount; r++) {
   210→   210→    const at = Math.min(deck.length, Math.round(((r + 1) * (count + roleCount)) / (roleCount + 1)) - 1 + r)
   211→   211→    deck.splice(Math.max(0, at), 0, { kind: "role", index: r })
   212→   212→  }
   213→   213→  const out: any[][] = []
   214→   214→  for (let c = 0; c < cols; c++) out.push([])
   215→   215→  deck.forEach((d, i) => out[i % cols].push({ kind: d.kind, index: d.index, dup: false }))
   216→   216→  if (!deck.length) return out
   217→   217→  // Pad from the far end of the column furthest away, so a repeat never sits
   218→   218→  // level with the same face in the next column.
   219→   219→  for (let c = 0; c < cols; c++) {
   220→   220→    let from = (c + Math.floor(cols / 2) + (cols % 2 ? 1 : 0)) % cols
   221→   221→    if (cols === 1) from = 0
   222→   222→    const pool = out[from].filter((e) => !e.dup).slice().reverse()
   223→   223→    const rest = deck.map((d) => ({ kind: d.kind, index: d.index, dup: false })).reverse()
   224→   224→    const candidates = pool.concat(rest)
   225→   225→    let k = 0
   226→   226→    let guard = 0
   227→   227→    while (out[c].length < minPerColumn && guard < candidates.length * 3) {
   228→   228→      const d = candidates[k % candidates.length]
   229→   229→      k++
   230→   230→      guard++
   231→   231→      const clash = out[c].some((e) => e.kind === d.kind && e.index === d.index)
   232→   232→      if (clash && guard < candidates.length) continue
   233→   233→      out[c].push({ kind: d.kind, index: d.index, dup: true })
   234→   234→    }
   235→   235→  }
   236→   236→  return out
   237→   237→}

   405→   405→function bringIntoView(card: HTMLElement) {
   406→   406→  const frame = card.closest(".dtr-wall, .dtr-strip") as HTMLElement | null
   407→   407→  const track = card.closest(".dtr-track, .dtr-strip-track") as HTMLElement | null
   408→   408→  if (!frame || !track) return
   409→   409→  const f = frame.getBoundingClientRect()
   410→   410→  const r = card.getBoundingClientRect()
   411→   411→  const vertical = track.classList.contains("dtr-track")
   412→   412→  // Start from where the track is right now (it may still be easing back).
   413→   413→  const now = getComputedStyle(track).translate
   414→   414→  const parts = !now || now === "none" ? ["0", "0"] : now.split(" ")
   415→   415→  let x = parseFloat(parts[0]) || 0
   416→   416→  let y = parseFloat(parts[1] || "0") || 0
   417→   417→  const m = 24
   418→   418→  if (vertical) {
   419→   419→    if (r.top < f.top + m) y += f.top + m - r.top
   420→   420→    else if (r.bottom > f.bottom - m) y -= r.bottom - (f.bottom - m)
   421→   421→  } else {
   422→   422→    if (r.left < f.left + m) x += f.left + m - r.left
   423→   423→    else if (r.right > f.right - m) x -= r.right - (f.right - m)
   424→   424→  }
   425→   425→  track.style.translate = Math.round(x) + "px " + Math.round(y) + "px"
   426→   426→}
